cmake_minimum_required(VERSION 3.20)
project(wdr_host LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 20)
set(XROBOT_MODULES_DIR "${CMAKE_CURRENT_SOURCE_DIR}/Modules")
set(WDR_PROTOCOL_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../Modules/HFUT-WDR/HostData")
set(LIBXR_SYSTEM linux CACHE STRING "" FORCE)
set(LIBXR_DRIVER linux CACHE STRING "" FORCE)
set(LIBXR_DEFAULT_SCALAR float CACHE STRING "" FORCE)
option(WDR_HOST_ROS "编译ROS2共享消息适配" OFF)
add_subdirectory(../Middlewares/Third_Party/LibXR libxr)
add_executable(wdr_host User/app_main.cpp)
target_include_directories(wdr_host PRIVATE User)
target_link_libraries(wdr_host PRIVATE xr)
if(WDR_HOST_ROS)
  find_package(ament_cmake REQUIRED)
  find_package(rclcpp REQUIRED)
  find_package(wdr_interfaces REQUIRED)
  target_compile_definitions(wdr_host PRIVATE WDR_HOST_ROS=1)
  ament_target_dependencies(wdr_host PUBLIC rclcpp)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(clock_tests tests/clock_tests.cpp)
  target_link_libraries(clock_tests PRIVATE xr)
  add_test(NAME clock_sync COMMAND clock_tests)
  add_executable(link_tests tests/link_tests.cpp)
  target_link_libraries(link_tests PRIVATE xr util)
  add_test(NAME link_pty COMMAND link_tests)
  set_tests_properties(link_pty PROPERTIES TIMEOUT 10)
  if(WDR_HOST_ROS)
    add_executable(ros_tests tests/ros_tests.cpp)
    target_link_libraries(ros_tests PRIVATE xr)
    ament_target_dependencies(ros_tests PUBLIC rclcpp wdr_interfaces)
    add_test(NAME ros_topics COMMAND ros_tests)
    set_tests_properties(ros_topics PROPERTIES TIMEOUT 8 ENVIRONMENT "ROS_DOMAIN_ID=73")
  endif()
endif()
